一、三种基本程序结构

顺序代码自上而下逐行执行，是最基本的结构。

分支if / else if / else、switch —— 按条件选择执行路径。

循环for、while、do...while、for...in、for...of —— 重复执行一段代码。

// 顺序结构
var fzg_a = 1;
var fzg_b = 2;
var fzg_sum = fzg_a + fzg_b;   // 3

// 分支结构
if (fzg_sum > 2) {
  // 条件成立走这里
} else {
  // 条件不成立走这里
}

二、for 循环（已知次数首选）

// for(初始化; 条件; 更新) { 循环体 }
var fzg_total = 0;
for (var fzg_i = 1; fzg_i <= 10; fzg_i++) {
  fzg_total += fzg_i;      // 累加 1~10
}

三、while 与 do...while 循环

// while：先判断后执行，条件不成立一次都不执行
var fzg_n = 1;
while (fzg_n <= 5) { fzg_n *= 2; }   // 1→2→4→8

// do...while：先执行后判断，循环体至少执行一次
var fzg_m = 100;
do {
  fzg_m++;          // 即使条件不成立也会先执行一次
} while (fzg_m < 100);

四、for...in 与 for...of

// for...in：遍历对象的键（属性名）
var fzg_stu = { fzg_name: "冯子格", fzg_age: 20, fzg_no: "228" };
for (var fzg_key in fzg_stu) { /* fzg_key 是属性名 */ }

// for...of：遍历数组的值
var fzg_arr = [10, 20, 30];
for (var fzg_val of fzg_arr) { /* fzg_val 是元素值 */ }

五、break 与 continue

// break：立即终止整个循环
// continue：跳过本次循环，进入下一次
for (var fzg_i = 1; fzg_i <= 10; fzg_i++) {
  if (fzg_i === 3) continue;   // 跳过 3
  if (fzg_i === 7) break;      // 到 7 结束
}

// 外层控制行，内层控制列
for (var fzg_i = 1; fzg_i <= 9; fzg_i++) {
  for (var fzg_j = 1; fzg_j <= fzg_i; fzg_j++) {
    // fzg_j × fzg_i = 积
  }
}
